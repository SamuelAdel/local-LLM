WITH OrderRevenue AS (
    SELECT
        oi.product_id,
        SUM(oi.quantity * oi.unit_price * (1 - oi.discount)) AS revenue
    FROM Order_Items AS oi
    GROUP BY
        oi.product_id
)
SELECT
    c.category_name,
    SUM(or.revenue) AS total_revenue
FROM Categories AS c
JOIN Products AS p
    ON c.category_id = p.category_id
JOIN OrderRevenue AS or
    ON p.product_id = or.product_id
GROUP BY
    c.category_name;

WITH CustomerOrderTotals AS (

    o.customer_id,
    SUM(oi.quantity * oi.unit_price * (1 - oi.discount)) AS total_spent
  FROM Orders AS o
  JOIN Order_Items AS oi
    ON o.order_id = oi.order_id
  GROUP BY
    o.customer_id
)

WITH EmployeeOrderValues AS (
    SELECT
        o.employee_id,
        SUM(o.total_amount) AS total_order_value
    FROM Orders o
    JOIN Employees e ON o.employee_id = e.employee_id
    GROUP BY
        o.employee_id
)
SELECT
    e.first_name,
    e.last_name,
    AVG(elv.total_order_value) AS average_order_value
FROM Employees e

    e.employee_id,
    e.first_name,
